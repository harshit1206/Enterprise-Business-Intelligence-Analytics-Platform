ENTERPRISE BUSINESS INTELLIGENCE & ANALYTICS PLATFORM
Power BI DAX Measures
Dataset: E-Commerce Sales Data

============================================================
-- 1. BASE / CORE BUSINESS MEASURES --
============================================================

Total Sales :=
SUM ( FactSales[Sales] )

Total Profit :=
SUM ( FactSales[Profit] )

Total Orders :=
DISTINCTCOUNT ( FactSales[Order ID] )

Total Quantity :=
SUM ( FactSales[Quantity] )

Total Customers :=
DISTINCTCOUNT ( FactSales[CustomerKey] )

Average Order Value :=
DIVIDE ( [Total Sales], [Total Orders] )

Profit Margin % :=
DIVIDE ( [Total Profit], [Total Sales] )

Average Unit Price :=
AVERAGE ( FactSales[Unit Price] )

Average Discount :=
AVERAGE ( FactSales[Discount] )

Total Discount Value :=
SUMX (
    FactSales,
    FactSales[Quantity] * FactSales[Unit Price] * FactSales[Discount]
)

-- 2. ORDER / SALES ANALYTICS MEASURES --

Orders per Customer :=
DIVIDE ( [Total Orders], [Total Customers] )

Items per Order :=
DIVIDE ( [Total Quantity], [Total Orders] )

Sales per Customer :=
DIVIDE ( [Total Sales], [Total Customers] )

Profit per Customer :=
DIVIDE ( [Total Profit], [Total Customers] )

Profit per Order :=
DIVIDE ( [Total Profit], [Total Orders] )

Sales per Item :=
DIVIDE ( [Total Sales], [Total Quantity] )

Quantity per Customer :=
DIVIDE ( [Total Quantity], [Total Customers] )


============================================================
-- 3. TIME INTELLIGENCE MEASURES --
============================================================

Sales YTD :=
TOTALYTD (
    [Total Sales],
    DimDate[Date]
)

Sales MTD :=
TOTALMTD (
    [Total Sales],
    DimDate[Date]
)

Sales QTD :=
TOTALQTD (
    [Total Sales],
    DimDate[Date]
)

Sales LY :=
CALCULATE (
    [Total Sales],
    SAMEPERIODLASTYEAR ( DimDate[Date] )
)

Profit YTD :=
TOTALYTD (
    [Total Profit],
    DimDate[Date]
)

Profit LY :=
CALCULATE (
    [Total Profit],
    SAMEPERIODLASTYEAR ( DimDate[Date] )
)

Sales YoY :=
[Total Sales] - [Sales LY]

Sales YoY % :=
DIVIDE (
    [Total Sales] - [Sales LY],
    [Sales LY]
)

Profit YoY :=
[Total Profit] - [Profit LY]

Profit YoY % :=
DIVIDE (
    [Total Profit] - [Profit LY],
    [Profit LY]
)

Rolling 30D Sales :=
CALCULATE (
    [Total Sales],
    DATESINPERIOD (
        DimDate[Date],
        MAX ( DimDate[Date] ),
        -30,
        DAY
    )
)

Rolling 90D Sales :=
CALCULATE (
    [Total Sales],
    DATESINPERIOD (
        DimDate[Date],
        MAX ( DimDate[Date] ),
        -90,
        DAY
    )
)

Rolling 30D Profit :=
CALCULATE (
    [Total Profit],
    DATESINPERIOD (
        DimDate[Date],
        MAX ( DimDate[Date] ),
        -30,
        DAY
    )
)

============================================================
-- 9. PROFITABILITY / DISCOUNT ANALYSIS --
============================================================

Sales Before Discount :=
SUMX (
    FactSales,
    FactSales[Quantity] * FactSales[Unit Price]
)

Discount Amount :=
[Sales Before Discount] - [Total Sales]

Discount % of Gross Sales :=
DIVIDE (
    [Discount Amount],
    [Sales Before Discount]
)

Profit per Unit :=
DIVIDE (
    [Total Profit],
    [Total Quantity]
)

Sales vs Profit Gap :=
[Total Sales] - [Total Profit]

Low Margin Sales :=
CALCULATE (
    [Total Sales],
    FILTER (
        FactSales,
        DIVIDE ( FactSales[Profit], FactSales[Sales] ) < 0.10
    )
)
